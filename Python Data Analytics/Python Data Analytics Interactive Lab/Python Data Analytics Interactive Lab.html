import pandas as pd import numpy as np # Create DataFrame with missing values and duplicates data = { 'CustomerID': ['C001', 'C002', 'C001', 'C003', 'C004'], 'Name': ['John Smith', 'Jane Doe', 'John Smith', None, 'Alice Johnson'], 'Email': ['john@email.com', 'jane@email.com', 'john@email.com', 'bob@email.com', 'alice@email.com'], 'PurchaseDate': ['2024-01-15', '2024/01/16', '2024-01-15', 'Jan 17, 2024', '2024-01-18'], 'Amount': ['$150.00', '$225.50', '$150.00', '$300', '$175.75'], 'Status': ['Active', 'Active', 'Active', 'Inactive', 'active'] } df = pd.DataFrame(data) print("Original messy data:") print(df) # Check for missing values print("\nMissing values:") print(df.isnull().sum()) # Remove duplicate rows df_clean = df.drop_duplicates() print("\nAfter removing duplicates:") print(df_clean.shape) # Fill missing values df_clean['Name'] = df_clean['Name'].fillna('Unknown') # Standardize Status column df_clean['Status'] = df_clean['Status'].str.capitalize() # Clean Amount column (remove $ and convert to float) df_clean['Amount'] = df_clean['Amount'].str.replace('$', '').astype(float) # Standardize PurchaseDate (convert to datetime) df_clean['PurchaseDate'] = pd.to_datetime(df_clean['PurchaseDate'], errors='coerce') print("\nCleaned DataFrame:") print(df_clean) print("\nDataFrame info after cleaning:") print(df_clean.info())
🔄 Modifying and Replacing Values

Modify DataFrame values, handle outliers, and replace data.

modifying.py
import pandas as pd import numpy as np # Create sample DataFrame with some anomalies data = { 'Student': ['Alice', 'Bob', 'Charlie', 'Diana', 'Eve'], 'Score': [95, 87, 150, 92, -10], # 150 and -10 are anomalies 'Grade': ['A', 'B', 'A', 'A', 'F'], 'Attendance': [95, 88, None, 92, 70] # One missing value } df = pd.DataFrame(data) print("Original DataFrame:") print(df) # Check for anomalous values in Score column score_mean = df['Score'].mean() score_std = df['Score'].std() z_scores = (df['Score'] - score_mean) / score_std print("\nZ-scores for Score column:") print(z_scores) # Identify outliers (z-score > 2 or < -2) outliers = df[abs(z_scores) > 2] print("\nOutliers detected:") print(outliers) # Replace outliers with median using .loc score_median = df['Score'].median() df.loc[abs(z_scores) > 2, 'Score'] = score_median # Fill missing attendance with mean attendance_mean = df['Attendance'].mean() df['Attendance'] = df['Attendance'].fillna(attendance_mean) # Create a new column based on conditions df['Performance'] = np.where(df['Score'] >= 90, 'Excellent', np.where(df['Score'] >= 80, 'Good', np.where(df['Score'] >= 70, 'Average', 'Poor'))) # Modify Grade column based on Performance df.loc[df['Performance'] == 'Excellent', 'Grade'] = 'A' df.loc[df['Performance'] == 'Good', 'Grade'] = 'B' print("\nDataFrame after modifications:") print(df) # Check for null values print("\nNull values after cleaning:") print(df.isnull().sum())

import pandas as pd import numpy as np import matplotlib.pyplot as plt import seaborn as sns print("CHESS GAME ANALYSIS PROJECT") print("=" * 50) # Create simulated chess games dataset np.random.seed(42) n_games = 1000 data = { 'game_id': range(1, n_games + 1), 'rated': np.random.choice([True, False], n_games, p=[0.7, 0.3]), 'turns': np.random.randint(10, 150, n_games), 'victory_status': np.random.choice(['mate', 'resign', 'outoftime', 'draw'], n_games, p=[0.3, 0.4, 0.2, 0.1]), 'winner': np.random.choice(['white', 'black', 'draw'], n_games, p=[0.45, 0.45, 0.1]), 'white_rating': np.random.normal(1500, 300, n_games).astype(int), 'black_rating': np.random.normal(1500, 300, n_games).astype(int), 'opening_name': np.random.choice([ "Queen's Pawn Game", "Sicilian Defense", "French Defense", "Caro-Kann Defense", "Italian Game", "Ruy Lopez", "King's Indian Defense", "English Opening" ], n_games), 'time_control': np.random.choice(['600', '300+5', '180+2', '900'], n_games) } # Create rating difference and ensure it affects win probability data['rating_diff'] = data['white_rating'] - data['black_rating'] win_prob = 1 / (1 + 10 ** (-data['rating_diff'] / 400)) # Adjust winner based on rating difference probability adjusted_winners = [] for prob in win_prob: if np.random.random() < prob * 0.8: # White advantage adjusted_winners.append('white') elif np.random.random() < (1 - prob) * 0.8: # Black advantage adjusted_winners.append('black') else: adjusted_winners.append('draw') data['winner'] = adjusted_winners chess_df = pd.DataFrame(data) print("Dataset Overview:") print("Number of games:", len(chess_df)) print("Columns:", chess_df.columns.tolist()) print("\nFirst 5 games:") print(chess_df.head()) print("\nBasic Statistics:") print(chess_df.describe()) # 1.
